#!/usr/bin/env python3
"""Картинки новости в webp: python3 convert-images.py <папка-с-исходниками> <public/news/slug>

cover.* сохраняется шириной 2000 px, остальные — 1440 px (колонка текста 720 px на экране 2x), quality 80.
Повреждённый (недокачанный) файл не конвертируется: скрипт сообщает о нём и завершается с ошибкой.
"""
import sys
from pathlib import Path

from PIL import Image

COVER_WIDTH = 2000
IMAGE_WIDTH = 1440


def main() -> int:
    if len(sys.argv) != 3:
        print(__doc__)
        return 1

    source, target = Path(sys.argv[1]), Path(sys.argv[2])
    target.mkdir(parents=True, exist_ok=True)
    broken = []

    for path in sorted(source.iterdir()):
        if path.suffix.lower() not in {".png", ".jpg", ".jpeg", ".webp"}:
            continue

        try:
            image = Image.open(path)
            image.load()
        except OSError as error:
            broken.append(f"{path.name}: {error}")
            continue

        width = COVER_WIDTH if path.stem == "cover" else IMAGE_WIDTH
        image = image.convert("RGB")

        if image.width > width:
            image = image.resize((width, round(image.height * width / image.width)), Image.LANCZOS)

        out = target / f"{path.stem}.webp"
        image.save(out, "WEBP", quality=80, method=6)
        print(f"{out}: {image.width}×{image.height}, {out.stat().st_size // 1024} КБ")

    for line in broken:
        print(f"Повреждён, пропущен — {line}", file=sys.stderr)

    return 1 if broken else 0


if __name__ == "__main__":
    sys.exit(main())
