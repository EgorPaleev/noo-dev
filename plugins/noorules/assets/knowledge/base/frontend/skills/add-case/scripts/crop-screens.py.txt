#!/usr/bin/env python3
"""Обрезка одноцветной подложки вокруг экранов и сохранение в webp для слайдера кейса.

Использование: python3 crop-screens.py <папка-назначения> <файл>=<имя> [<файл>=<имя> ...]
Пример:        python3 crop-screens.py client-sovpa/public/cases/tech-expo-2027 shot1.png=matching shot2.png=form
Печатает размеры готовых файлов: их нужно записать в width/height экрана.
"""
import os
import sys

from PIL import Image, ImageChops

MAX_SIDE = 1800


def crop(src: str, dst: str) -> None:
    image = Image.open(src).convert("RGB")
    background = Image.new("RGB", image.size, image.getpixel((5, 5)))
    box = ImageChops.difference(image, background).convert("L").point(lambda value: 255 if value > 18 else 0).getbbox()
    result = image.crop(box) if box else image
    result.thumbnail((MAX_SIDE, MAX_SIDE))
    result.save(dst, quality=82, method=6)
    print(f"{dst}: {result.size[0]}×{result.size[1]}, {os.path.getsize(dst) // 1024} КБ")


if __name__ == "__main__":
    if len(sys.argv) < 3:
        sys.exit(__doc__)
    out_dir = sys.argv[1]
    os.makedirs(out_dir, exist_ok=True)
    for pair in sys.argv[2:]:
        source, name = pair.split("=", 1)
        crop(source, os.path.join(out_dir, f"{name}.webp"))
